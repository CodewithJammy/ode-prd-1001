// Handle option selection document.querySelectorAll('.option-btn').forEach(btn => { btn.addEventListener('click', () => { let qid = btn.dataset.qid; // Reset styles for all options of this question document.querySelectorAll(`[data-qid="${qid}"]`).forEach(b => b.classList.remove('btn-primary','text-white')); // Highlight chosen btn.classList.add('btn-primary','text-white'); // Save choice document.getElementById('chosen'+qid).value = btn.dataset.value; }); }); function submitTest() { let total = document.querySelectorAll('#questions .card').length; let correct = 0; let wrong = 0; let resultsHTML = ""; document.querySelectorAll('#questions .card').forEach(card => { let qid = card.querySelector('input[id^=ans]').id.replace('ans',''); let chosen = document.getElementById('chosen'+qid).value; let correctAns = document.getElementById('ans'+qid).value; if (chosen === correctAns) { correct++; } else if (chosen) { wrong++; } let qText = card.querySelector('p').innerHTML; // Build result card resultsHTML += `

${qText}

Your Answer: ${chosen || 'Not Attempted'}

Correct Answer: ${correctAns}

`; }); let accuracy = ((correct / total) * 100).toFixed(0); let scoreBox = document.getElementById('scoreBox'); scoreBox.innerHTML = `

${accuracy}% Accuracy

${correct} Correct, ${wrong} Wrong

${accuracy>=80 ? '✅ Passed' : '❌ Failed — Needs 80%'}

`; document.getElementById('resultsContent').innerHTML = resultsHTML; let resultsTab = new bootstrap.Tab(document.querySelector('#results-tab')); resultsTab.show(); } function retryTest() { document.querySelectorAll('.option-btn').forEach(btn => btn.classList.remove('btn-primary','text-white')); document.querySelectorAll('[id^=chosen]').forEach(h => h.value = ""); document.getElementById('scoreBox').innerHTML = ""; let questionsTab = new bootstrap.Tab(document.querySelector('#questions-tab')); questionsTab.show(); }
